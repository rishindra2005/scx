\begin{abstract}
Asymmetric multiprocessor architectures, combining high-throughput Performance (P) cores and energy-efficient Efficiency (E) cores with disparate compute capacities, present fundamental challenges for general-purpose operating system schedulers. Existing kernel schedulers rely heavily on empirical heuristics that lack formal mathematical characterization and struggle under mixed latency-critical and compute-intensive workloads. In this paper, we develop the rigorous mathematical foundations and scheduling theory of OPTIMA, an extensible task-scheduling framework implemented via Linux \texttt{sched\_ext} eBPF. 

We formalize the asymmetric scheduling problem under a dual objective: minimizing total weighted completion time $\sum w_j C_j$ while providing latency-critical prioritization and bounded queuing delay under non-overloaded regimes. We rigorously prove that Weighted Shortest Processing Time (WSPT) sequencing via Smith's Rule is optimal for isolated, single-core non-preemptive queues at time zero ($1 || \sum w_j C_j$). We explicitly acknowledge that the generalized online problem with arbitrary release dates, preemption, and asymmetric parallel machines ($R | r_j, \text{pmtn} | \sum w_j C_j$) is strongly NP-hard; accordingly, we cite the classical Kawaguchi-Kyan bound ($\alpha \le \frac{1+\sqrt{2}}{2} \approx 1.207$) strictly as an identical-machine reference bound and characterize OPTIMA's density-based virtual deadline list scheduling as an online approximation heuristic. Using the Hardy-Littlewood-P\'olya Rearrangement Inequality and a continuous fluid progress rate formulation, we prove that matching tasks with higher execution density to Performance cores maximizes the instantaneous priority-weighted progress rate.

Furthermore, we formalize an Online Admissible Preemption Filter (OAPF) based on a dimensionally consistent weighted one-step dispatch delay proxy $\Delta\mathrm{DelayProxy}(c) = w_{\mathrm{curr}}(c) \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}}) - w_{\mathrm{wake}} \cdot R(c)$. We prove that this greedy filter enforces priority inversion safety, eliminates context-switch thrashing, and guarantees strict local one-step dispatch delay reduction. Finally, using a quadratic Lyapunov drift formulation with an exact cross-term expansion, we prove that OPTIMA's adaptive quantum scaling and hierarchical work-stealing maintain strong queue stability ($\limsup \frac{1}{T} \sum \mathbb{E}[q_k(t)] < \infty$) under all strictly admissible workloads ($\sum \lambda_k \bar{p}_k \le C_{\text{total}} - \epsilon_0$). By bounding virtual deadline advancement ($v_j \le t + 2 \tau_{\max}$ with increments $\Delta v_j \le \tau_{\max}$), we reconcile strict priority queuing with guaranteed starvation freedom for every admitted task. Complete algorithmic specifications in kernel eBPF hot-paths are provided.
\end{abstract}

\section{System Model and Problem Formulation}
\label{sec:system_model}

\subsection{Heterogeneous Multiprocessor Architecture}
We model a heterogeneous asymmetric chip-multiprocessor (AMP) as a set of $M$ discrete computing cores denoted by:
\begin{equation}
\mathcal{M} = \mathcal{P} \cup \mathcal{E}, \quad \mathcal{P} \cap \mathcal{E} = \emptyset
\end{equation}
where $\mathcal{P} = \{C_1, C_2, \dots, C_{M_P}\}$ represents the partition of $M_P = |\mathcal{P}|$ Performance cores (P-cores), and $\mathcal{E} = \{C_{M_P+1}, C_{M_P+2}, \dots, C_{M_P+M_E}\}$ represents the partition of $M_E = |\mathcal{E}|$ Efficiency cores (E-cores), yielding total core count $M = M_P + M_E$.

Each core $C_k \in \mathcal{M}$ is characterized by an intrinsic, time-invariant computational capacity (effective instruction processing frequency) $c_k \in \mathbb{R}_{> 0}$, defined as:
\begin{equation}
c_k = \begin{cases} 
c_P, & \forall C_k \in \mathcal{P} \\ 
c_E, & \forall C_k \in \mathcal{E} 
\end{cases}
\end{equation}
where $c_P > c_E > 0$. We define the asymmetric processing speedup ratio as:
\begin{equation}
\mu \triangleq \frac{c_P}{c_E} > 1
\end{equation}
The total aggregate computational capacity rate of the multiprocessor system is:
\begin{equation}
C_{\mathrm{total}} \triangleq \sum_{k=1}^M c_k = M_P c_P + M_E c_E
\end{equation}

The memory and cache subsystem is structured hierarchically: P-cores in $\mathcal{P}$ share a high-bandwidth coherent L3 cache domain (CCX), while E-cores in $\mathcal{E}$ reside in an independent L3 cache domain. Cores may optionally support Simultaneous Multithreading (SMT), where physical core $C_k$ exposes two hardware thread contexts sharing execution units and L1/L2 caches.

\subsection{Dynamic Online Task Model}
The workload submitted to the operating system is represented as a dynamic online sequence of tasks:
\begin{equation}
\mathcal{J} = \{J_1, J_2, \dots, J_n, \dots\}
\end{equation}
Each task $J_j \in \mathcal{J}$ is formally specified by a 5-tuple:
\begin{equation}
J_j = \left( r_j, \, p_j, \, w_j, \, d_j, \, \mathcal{T}_j \right)
\end{equation}
where:
\begin{enumerate}
    \item $r_j \in \mathbb{R}_{\ge 0}$ denotes the release (arrival or wakeup) timestamp of task $J_j$.
    \item $p_j \in \mathbb{R}_{> 0}$ represents the nominal processing requirement (computational work volume in reference execution cycles or execution duration measured on a baseline core of capacity $c_E$). In the kernel implementation, $p_j$ is estimated via an Exponentially Weighted Moving Average (EWMA) of past burst durations.
    \item $w_j \in [w_{\min}, w_{\max}] \subset \mathbb{R}_{> 0}$ denotes the task priority weight, derived from POSIX nice levels via $w_j = 100 \times 1.25^{-\mathrm{nice}_j}$, with $w_{\min} \ge 10$.
    \item $d_j \in \mathbb{R}_{> 0} \cup \{\infty\}$ is the target completion deadline. Tasks are categorized into latency-critical/interactive tasks $\mathcal{J}_{\mathrm{RT}} \triangleq \{J_j \in \mathcal{J} \mid d_j < \infty\}$ and compute-bound batch tasks $\mathcal{J}_{\mathrm{Batch}} \triangleq \{J_j \in \mathcal{J} \mid d_j = \infty\}$.
    \item $\mathcal{T}_j \in \{\text{Interactive}, \text{Batch}\}$ designates the behavioral execution classification.
\end{enumerate}

\begin{definition}[Task Density]
\label{def:density}
For any task $J_j \in \mathcal{J}$, its task density $\rho_j$ is defined as the ratio of its priority weight to its nominal execution requirement:
\begin{equation}
\rho_j \triangleq \frac{w_j}{p_j}
\end{equation}
representing the marginal priority weight delivered per unit of computational work.
\end{definition}

\subsection{Execution Capacity Scaling}
The physical execution duration of task $J_j$ when dispatched to processor core $C_k \in \mathcal{M}$ depends inversely on the processing capacity $c_k$:
\begin{equation}
p_{j,k} = \frac{p_j}{c_k} = \begin{cases} 
\dfrac{p_j}{c_P}, & C_k \in \mathcal{P} \\[8pt] 
\dfrac{p_j}{c_E} = \mu \cdot \dfrac{p_j}{c_P}, & C_k \in \mathcal{E} 
\end{cases}
\end{equation}
Thus, executing task $J_j$ on a Performance core accelerates execution by a factor of $\mu$ relative to execution on an Efficiency core.

\subsection{Problem Formulation and Real-Time Characterization}
Let $S_j \ge r_j$ and $C_j$ denote the scheduling start time and final completion time of task $J_j$ under schedule $\mathcal{S}$, respectively. If task $J_j$ undergoes preemptive execution across time slots, its total completed work satisfies:
\begin{equation}
\sum_{k \in \mathcal{M}} c_k \int_{S_j}^{C_j} \mathbf{1}_{\{J_j \text{ runs on } C_k \text{ at time } t\}} \, dt = p_j
\end{equation}

The primary objective is to minimize the total weighted completion time across all tasks:
\begin{align}
\min_{\mathcal{S}} \quad & \Phi(\mathcal{S}) \triangleq \sum_{J_j \in \mathcal{J}} w_j C_j(\mathcal{S}) \label{eq:obj} \\
\text{subject to} \quad & S_j \ge r_j, \quad \forall J_j \in \mathcal{J} \label{eq:causality_constraint} \\
& \sum_{J_j \in \mathcal{J}} \mathbf{1}_{\{J_j \text{ on } C_k \text{ at } t\}} \le 1, \quad \forall C_k \in \mathcal{M}, \; t \ge 0 \label{eq:exclusive_constraint}
\end{align}
Constraint~\eqref{eq:causality_constraint} enforces causality, and Constraint~\eqref{eq:exclusive_constraint} guarantees non-overlapping single-task execution per physical processing core.

\begin{remark}[Real-Time Guarantees and Admission Control]
\label{rem:realtime_softening}
In an open, general-purpose operating system without hard admission control, arbitrary processes may arrive concurrently without reservation contracts. Under arbitrary adversarial overload ($\sum_{j \in \mathcal{J}_{\mathrm{RT}}} p_j / (d_j - r_j) > M_P$), it is mathematically impossible for any online scheduler to guarantee hard real-time deadlines ($C_j \le d_j$) for all tasks. Consequently, OPTIMA formulates real-time support as:
\begin{enumerate}
    \item \textbf{Latency-Critical Prioritization:} Latency-critical tasks are prioritized via zero-lag local dispatch and minimal virtual deadlines ($v_j = t$), ensuring immediate dispatch to Performance cores when capacity is available.
    \item \textbf{Bounded Queuing Delay under Non-Overload:} Under non-overloaded regimes where the latency-critical workload utilization satisfies $\sum_{j \in \mathcal{J}_{\mathrm{RT}}} \frac{p_j}{d_j - r_j} \le U_{\mathrm{RT}} < M_P$, OPTIMA provides bounded queuing delay and rapid dispatch onto Performance cores without priority inversion.
    \item \textbf{Graceful Degradation:} Under system overload, OPTIMA provides graceful degradation by minimizing total weighted completion time $\sum w_j C_j$, preventing unbounded starvation.
\end{enumerate}
\end{remark}

\section{WSPT Scheduling: Single-Core Optimality, Computational Complexity, and Asymmetric Partitioning}
\label{sec:theorem1}

\subsection{Smith's Rule Optimality on Single-Core Dedicated Queues}
We begin by establishing the formal scope within which Weighted Shortest Processing Time (WSPT) sequencing is strictly optimal: an isolated single-core queue with simultaneously available independent jobs.

\begin{lemma}[Optimality of Single-Core Smith's Rule ($1 || \sum w_j C_j$)]
\label{lem:smith_single}
Let $\mathcal{J}_k = \{J_1, J_2, \dots, J_n\}$ be a set of $n$ independent tasks available simultaneously at time $t_0$ ($r_j = t_0, \forall j$) on a single isolated processor core $C_k$ with compute capacity $c_k$. Any non-preemptive schedule that orders tasks in non-increasing order of task density:
\begin{equation}
\label{eq:smith_order}
\rho_{(1)} \ge \rho_{(2)} \ge \dots \ge \rho_{(n)}
\end{equation}
where $\rho_j = w_j / p_j$, strictly minimizes the total weighted completion time $\sum_{j=1}^n w_j C_j$.
\end{lemma}

\begin{proof}
We proceed via an adjacent pairwise interchange argument. Assume for contradiction that there exists an optimal schedule $\mathcal{S}^*$ that does not satisfy the non-increasing density ordering~\eqref{eq:smith_order}.

Since $\mathcal{S}^*$ does not follow the monotonic density ordering everywhere, there must exist at least two adjacent tasks, say $J_a$ immediately followed by $J_b$, such that:
\begin{equation}
\label{eq:inv_cond}
\rho_a < \rho_b \iff \frac{w_a}{p_a} < \frac{w_b}{p_b} \iff w_a p_b < w_b p_a
\end{equation}
Let $t \ge t_0$ denote the start time of task $J_a$ under schedule $\mathcal{S}^*$. Under $\mathcal{S}^*$, task $J_a$ executes during $[t, \, t + p_{a,k})$, and task $J_b$ executes during $[t + p_{a,k}, \, t + p_{a,k} + p_{b,k})$. Their respective completion times are:
\begin{align}
C_a(\mathcal{S}^*) &= t + \frac{p_a}{c_k} \\
C_b(\mathcal{S}^*) &= t + \frac{p_a}{c_k} + \frac{p_b}{c_k}
\end{align}
The combined contribution of $J_a$ and $J_b$ to the objective $\Phi(\mathcal{S}^*)$ is:
\begin{align}
\Phi_{a,b}(\mathcal{S}^*) &= w_a C_a(\mathcal{S}^*) + w_b C_b(\mathcal{S}^*) \nonumber \\
&= w_a \left( t + \frac{p_a}{c_k} \right) + w_b \left( t + \frac{p_a + p_b}{c_k} \right) \nonumber \\
&= (w_a + w_b)t + \frac{1}{c_k} \Big( w_a p_a + w_b p_a + w_b p_b \Big)
\end{align}

Now construct an alternative schedule $\mathcal{S}'$ by interchanging the execution positions of $J_a$ and $J_b$, while preserving the execution positions of all tasks scheduled prior to time $t$ and all tasks scheduled after time $t + \frac{p_a + p_b}{c_k}$.

Under schedule $\mathcal{S}'$, task $J_b$ executes first during $[t, \, t + p_{b,k})$, followed by task $J_a$ during $[t + p_{b,k}, \, t + p_{b,k} + p_{a,k})$. Their completion times under $\mathcal{S}'$ become:
\begin{align}
C_b(\mathcal{S}') &= t + \frac{p_b}{c_k} \\
C_a(\mathcal{S}') &= t + \frac{p_b}{c_k} + \frac{p_a}{c_k}
\end{align}
The combined contribution of $J_a$ and $J_b$ under $\mathcal{S}'$ is:
\begin{align}
\Phi_{a,b}(\mathcal{S}') &= w_b C_b(\mathcal{S}') + w_a C_a(\mathcal{S}') \nonumber \\
&= w_b \left( t + \frac{p_b}{c_k} \right) + w_a \left( t + \frac{p_b + p_a}{c_k} \right) \nonumber \\
&= (w_a + w_b)t + \frac{1}{c_k} \Big( w_b p_b + w_a p_b + w_a p_a \Big)
\end{align}

For all tasks $J_j \notin \{J_a, J_b\}$, completion times are unaffected:
\begin{enumerate}
    \item Tasks completed before time $t$ finish before the interchange occurs, so $C_j(\mathcal{S}') = C_j(\mathcal{S}^*)$.
    \item Tasks scheduled after $J_a$ and $J_b$ begin at $t + \frac{p_b + p_a}{c_k} = t + \frac{p_a + p_b}{c_k}$. Because the total execution duration of $\{J_a, J_b\}$ is invariant under permutation, the start and completion times of all subsequent tasks remain identical: $C_j(\mathcal{S}') = C_j(\mathcal{S}^*)$ for all subsequent $j$.
\end{enumerate}

Now evaluate the cost difference $\Delta \Phi \triangleq \Phi(\mathcal{S}') - \Phi(\mathcal{S}^*)$:
\begin{align}
\Delta \Phi &= \Phi_{a,b}(\mathcal{S}') - \Phi_{a,b}(\mathcal{S}^*) \nonumber \\
&= \frac{1}{c_k} \Big( w_b p_b + w_a p_b + w_a p_a - w_a p_a - w_b p_a - w_b p_b \Big) \nonumber \\
&= \frac{1}{c_k} \Big( w_a p_b - w_b p_a \Big)
\end{align}
From the inversion condition~\eqref{eq:inv_cond}, $w_a p_b - w_b p_a < 0$. Since $c_k > 0$, it follows that:
\begin{equation}
\Delta \Phi = \frac{1}{c_k}(w_a p_b - w_b p_a) < 0 \implies \Phi(\mathcal{S}') < \Phi(\mathcal{S}^*)
\end{equation}
This contradicts the assumption that $\mathcal{S}^*$ is optimal. Since the permutation group $S_n$ is finite and generated by adjacent transpositions, any schedule can be transformed into non-increasing density order via a finite sequence of adjacent exchanges, with each exchange strictly decreasing the objective function until no inversions remain. Thus, Smith's density ordering is uniquely optimal for single-core non-preemptive queues at time zero.
\end{proof}

\subsection{Computational Complexity of Multiprocessor Scheduling and NP-Hardness}
\label{sec:np_hardness}
While Smith's Rule provides an exact, polynomial-time solution for the static single-machine problem $1 || \sum w_j C_j$, the generalized operating system scheduling problem on asymmetric multiprocessors is computationally intractable.

Using standard Graham notation $\alpha | \beta | \gamma$:
\begin{enumerate}
    \item \textbf{Release Dates on a Single Core ($1 | r_j | \sum w_j C_j$):} Introducing dynamic task release dates makes the single-core non-preemptive problem strongly NP-hard, as proven by Lenstra, Rinnooy Kan, and Brucker (1977).
    \item \textbf{Preemptive Scheduling with Release Dates ($1 | r_j, \text{pmtn} | \sum w_j C_j$):} Even when arbitrary preemption is permitted, minimizing total weighted completion time with dynamic arrivals remains strongly NP-hard, as proven by Lenstra, Rinnooy Kan, and Brucker (1977). (Bruno, Coffman, and Sethi, 1974, established NP-completeness for precedence-constrained task graphs on identical processors).
    \item \textbf{Parallel Machines ($P || \sum w_j C_j$):} Scheduling independent jobs without release dates on identical parallel machines is strongly NP-hard (Garey and Johnson, 1979).
    \item \textbf{Asymmetric / Unrelated Parallel Machines ($R | r_j, \text{pmtn} | \sum w_j C_j$):} Scheduling dynamic tasks on asymmetric machines with disparate core capacities ($c_P \ne c_E$), arbitrary release dates, and preemption is strongly NP-hard.
\end{enumerate}

\begin{proposition}[Computational Intractability of Global Multi-Core Minimization]
\label{prop:np_hard}
Finding a schedule that globally minimizes total weighted completion time $\sum w_j C_j$ on an asymmetric multiprocessor $\mathcal{M} = \mathcal{P} \cup \mathcal{E}$ with online release dates and preemption ($R | r_j, \text{pmtn} | \sum w_j C_j$) is strongly NP-hard. No online deterministic scheduler can guarantee an exact global minimum in polynomial time unless $\mathrm{P} = \mathrm{NP}$.
\end{proposition}

\subsection{Approximation Ratio Reference and Online List Scheduling}
In classical scheduling theory, Kawaguchi and Kyan (1986) analyzed the worst-case approximation performance of WSPT list scheduling:
\begin{theorem}[Kawaguchi-Kyan Reference Bound on Identical Machines]
\label{thm:kawaguchi_kyan}
For the offline scheduling of independent jobs on $M$ identical parallel processors without release dates ($P || \sum w_j C_j$), the WSPT list scheduling heuristic achieves an approximation ratio bounded by:
\begin{equation}
\frac{\Phi(\mathcal{S}_{\mathrm{WSPT}})}{\Phi(\mathcal{S}_{\mathrm{OPT}})} \le \frac{1 + \sqrt{2}}{2} \approx 1.207
\end{equation}
\end{theorem}

\begin{remark}[Scope and Application to Asymmetric Architectures]
The Kawaguchi-Kyan bound $\alpha \le 1.207$ strictly applies to \emph{identical parallel machines} ($P$) with simultaneous job availability at time zero. On asymmetric multiprocessors ($Q$ or $R$) with dynamic release dates $r_j$, the Kawaguchi-Kyan bound does not apply directly. 

Consequently, OPTIMA does \emph{not} claim global multi-core optimality via Smith's Rule, nor does it claim that the 1.207 bound holds unconditionally across asymmetric cores. Instead, OPTIMA employs density-based list scheduling and virtual deadline ordering as an \textbf{online approximation heuristic} inspired by the single-core optimality of Smith's Rule and the classical identical-machine bound.
\end{remark}

\subsection{Instantaneous Backlog Weight Depletion via the Rearrangement Inequality}
We now provide the theoretical justification for density-tiered assignment across asymmetric cores $\mathcal{M} = \mathcal{P} \cup \mathcal{E}$ with $c_P > c_E$.

\begin{theorem}[Maximization of Instantaneous Backlog Weight Depletion Rate]
\label{thm:rearrangement_depletion}
Let $\mathcal{M} = \mathcal{P} \cup \mathcal{E}$ be an asymmetric multiprocessor system with capacities $c_P > c_E$. At any instant $t$, let $\mathcal{Q}(t)$ denote the set of active uncompleted tasks, and let $W(t) \triangleq \sum_{J_j \in \mathcal{Q}(t)} w_j$ denote total remaining priority weight. 

Matching tasks with higher density $\rho_j = w_j / p_j$ to Performance cores strictly maximizes the instantaneous rate of remaining weight reduction $-\dot{W}(t)$.
\end{theorem}

\begin{proof}
Using the integral representation of total weighted completion time:
\begin{equation}
\sum_{J_j \in \mathcal{J}} w_j C_j = \int_0^\infty W(t) \, dt
\end{equation}
To heuristically minimize $\int_0^\infty W(t) \, dt$, a greedy online scheduler should maximize the instantaneous rate of weight decay $-\dot{W}(t)$ at every time $t$.

At any time $t$, let $J_{\sigma(k)}$ be the task currently executing on core $C_k \in \mathcal{M}$. The computational rate delivered to task $J_{\sigma(k)}$ is $c_k / p_{\sigma(k)}$ per second. Because task $J_{\sigma(k)}$ completes after executing $p_{\sigma(k)}$ work, the effective rate of weight retirement delivered by core $C_k$ is:
\begin{equation}
-\dot{W}_k(t) = w_{\sigma(k)} \cdot \frac{c_k}{p_{\sigma(k)}} = \rho_{\sigma(k)} \cdot c_k
\end{equation}
Summing over all $M$ cores, the total instantaneous rate of weight depletion is:
\begin{equation}
\label{eq:weight_depletion_inner_prod}
-\dot{W}(t) = \sum_{k=1}^M \rho_{\sigma(k)} c_k = \sum_{k \in \mathcal{P}} \rho_{\sigma(k)} c_P + \sum_{k \in \mathcal{E}} \rho_{\sigma(k)} c_E
\end{equation}

Now consider any two tasks $J_1$ and $J_2$ running concurrently, with $J_1$ assigned to a P-core ($c_P$) and $J_2$ assigned to an E-core ($c_E$). Suppose for contradiction that $\rho_1 < \rho_2$.
The weight depletion rate under this assignment $\mathcal{A}$ is:
\begin{equation}
R(\mathcal{A}) = \rho_1 c_P + \rho_2 c_E
\end{equation}
Interchanging the core assignments of $J_1$ and $J_2$ produces assignment $\mathcal{A}'$ with depletion rate:
\begin{equation}
R(\mathcal{A}') = \rho_2 c_P + \rho_1 c_E
\end{equation}
Computing the difference:
\begin{align}
R(\mathcal{A}') - R(\mathcal{A}) &= (\rho_2 c_P + \rho_1 c_E) - (\rho_1 c_P + \rho_2 c_E) \nonumber \\
&= (\rho_2 - \rho_1)(c_P - c_E)
\end{align}
Because $\rho_2 > \rho_1$ and $c_P > c_E$, we have $(\rho_2 - \rho_1)(c_P - c_E) > 0$, which implies $R(\mathcal{A}') > R(\mathcal{A})$.

More generally, by the Hardy-Littlewood-P\'olya Rearrangement Inequality, for any two real sequences sorted in non-increasing order:
\begin{align}
\rho_{(1)} \ge \rho_{(2)} \ge \dots \ge \rho_{(M)} \\
c_{(1)} \ge c_{(2)} \ge \dots \ge c_{(M)}
\end{align}
the inner product $\sum_{k=1}^M \rho_{\sigma(k)} c_k$ is strictly maximized if and only if the permutation $\sigma$ matches the monotonic ordering, i.e., $\sigma(k) = (k)$.
Because $c_k = c_P$ for all $k \in \mathcal{P}$ and $c_k = c_E$ for all $k \in \mathcal{E}$ with $c_P > c_E$, maximum weight depletion requires:
\begin{equation}
\min_{k \in \mathcal{P}} \rho_{\sigma(k)} \ge \max_{k' \in \mathcal{E}} \rho_{\sigma(k')}
\end{equation}
Therefore, allocating tasks with highest densities to Performance cores strictly maximizes the instantaneous rate of system backlog reduction.
\end{proof}

\begin{remark}[Instantaneous vs Global Optimality]
Maximizing $-\dot{W}(t)$ is an instantaneous (greedy) rate maximization at time $t$. Because future arrivals $r_j > t$ dynamically alter queue state, instantaneous rate maximization does not constitute a proof of global multi-stage offline optimality over $[0, \infty)$, but provides a principled mathematical basis for asymmetric core partitioning.
\end{remark}

\subsection{Density-Guided Domain Placement via Continuous Fluid Progress Rate}
When tasks wake up dynamically, full combinatorial reassignment across all cores is computationally prohibitive. OPTIMA formulates core allocation between Performance and Efficiency domains through a continuous fluid progress rate relaxation.

Under a fluid relaxation where active jobs make continuous progress, the instantaneous priority-weighted rate of progress across the multiprocessor is:
\begin{equation}
\label{eq:fluid_progress_rate}
\mathcal{R}(t) \triangleq \sum_{k=1}^M \rho_{\sigma(k)} c_k = \sum_{k \in \mathcal{P}} \rho_{\sigma(k)} c_P + \sum_{k \in \mathcal{E}} \rho_{\sigma(k)} c_E
\end{equation}
where $\rho_{\sigma(k)} = w_{\sigma(k)} / p_{\mathrm{est},\sigma(k)}$ is the execution density of the task executing on core $k$. By the Hardy-Littlewood-P\'olya Rearrangement Inequality (Theorem~\ref{thm:rearrangement_depletion}), $\mathcal{R}(t)$ is strictly maximized when tasks with higher execution density are assigned to faster Performance cores ($c_P > c_E$).

OPTIMA operationalizes this fluid principle directly in kernel eBPF as an $\mathcal{O}(1)$ domain placement policy:
\begin{itemize}
    \item High-priority latency-critical tasks ($w_j \ge w_{\mathrm{RT}}$) are unconditionally placed into the dedicated Performance queue (\texttt{DSQ\_PERF}).
    \item Long-running compute batch tasks ($p_{\mathrm{est},j} \ge 3\,\text{ms}$ or tagged as batch) have low execution density $\rho_j$ and are placed into the Efficiency domain queue (\texttt{DSQ\_EFF}), offloading sustained execution from P-cores.
    \item Short interactive tasks ($p_{\mathrm{est},j} < 3\,\text{ms}$) with high execution density are routed to the domestic shared Performance queue (\texttt{DSQ\_SHARED\_P}), preserving Performance core bandwidth for rapid interactive turnaround.
\end{itemize}
This eliminates uncoordinated thread racing while providing a deterministic $\mathcal{O}(1)$ kernel hot-path decision rule grounded in fluid rate maximization.

\section{Online Admissible Preemption Filtering (OAPF)}
\label{sec:theorem2}

\subsection{The Online Preemption Decision Dilemma}
In an asymmetric multiprocessing system, when a high-priority interactive or latency-critical task $J_{\mathrm{wake}} \in \mathcal{J}_{\mathrm{RT}}$ arrives at time $t$, all Performance cores $\mathcal{P}$ may be actively executing batch tasks. The scheduler faces an immediate online dilemma:
\begin{itemize}
    \item \textbf{Option 1 (Non-Preemptive Wait):} Enqueue $J_{\mathrm{wake}}$ and allow the running task to complete its allocated time quantum, avoiding context-switch overhead but incurring queue latency for $J_{\mathrm{wake}}$.
    \item \textbf{Option 2 (Preemptive Displacement):} Select an active P-core $c^* \in \mathcal{P}$, immediately preempt its currently running task $J_{\mathrm{curr}}(c^*)$, and schedule $J_{\mathrm{wake}}$ on core $c^*$.
\end{itemize}

Preemption is not cost-free: preempting a task introduces a context-switch and cache displacement overhead $C_{\mathrm{switch}} > 0$, while delaying the running task $J_{\mathrm{curr}}$ by the execution duration of $J_{\mathrm{wake}}$ plus switching overhead.

\begin{remark}[Nature of the Algorithmic Decision]
OPTIMA does \emph{not} explore an exponential, multi-stage Branch-and-Bound search tree over future schedules. Instead, OPTIMA implements an \textbf{Online Admissible Preemption Filter (OAPF)} / \textbf{Greedy Decision Rule} that evaluates candidate Performance cores $c \in \mathcal{P}$ in $\mathcal{O}(M_P)$ bounded kernel time upon each task wakeup.
\end{remark}

\subsection{Dimensionally Consistent Cost Formulation}
For each candidate Performance core $c \in \mathcal{P}$, let $J_{\mathrm{curr}}(c)$ denote the task executing on core $c$. Let:
\begin{itemize}
    \item $w_{\mathrm{curr}}(c)$ be the priority weight of running task $J_{\mathrm{curr}}(c)$.
    \item $t_{\mathrm{start}}(c)$ be the start timestamp of $J_{\mathrm{curr}}(c)$'s current time slice.
    \item $\tau(c)$ be the allocated quantum of $J_{\mathrm{curr}}(c)$.
    \item $R(c) \triangleq \max\Big(0, \, \tau(c) - \big(t - t_{\mathrm{start}}(c)\big)\Big)$ be the remaining quantum duration of $J_{\mathrm{curr}}(c)$ in time units.
    \item $w_{\mathrm{wake}}$ be the priority weight of waking task $J_{\mathrm{wake}}$.
    \item $p_{\mathrm{wake}}$ be the nominal execution work / quantum assigned to $J_{\mathrm{wake}}$ (in kernel: bounded by $\tau_{\min}$).
    \item $C_{\mathrm{switch}}$ be the context-switch and cache invalidation penalty in time units.
\end{itemize}

\begin{definition}[Weighted One-Step Dispatch Delay Proxy]
\label{def:deltacost}
Consider candidate core $c \in \mathcal{P}$ currently executing task $J_{\mathrm{curr}}(c)$ with remaining quantum $R(c) > 0$. Let waking task $J_{\mathrm{wake}}$ have weight $w_{\mathrm{wake}}$ and allocated time slice $\tau_{\mathrm{slice}} = \tau_{\min}$. If the scheduler chooses \textbf{Option 1 (Wait)}, waking task $J_{\mathrm{wake}}$ is delayed by $R(c)$, incurring weighted dispatch delay:
\begin{equation}
\mathrm{Delay}_{\mathrm{wait}}(c) = w_{\mathrm{wake}} \cdot R(c)
\end{equation}
If the scheduler chooses \textbf{Option 2 (Preempt)}, $J_{\mathrm{wake}}$ executes immediately with 0 delay. However, running task $J_{\mathrm{curr}}(c)$ is suspended during $J_{\mathrm{wake}}$'s slice ($\tau_{\mathrm{slice}}$ time units) plus context-switch overhead $C_{\mathrm{switch}}$, incurring weighted delay penalty:
\begin{equation}
\mathrm{Delay}_{\mathrm{preempt}}(c) = w_{\mathrm{curr}}(c) \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})
\end{equation}

The net differential in one-step weighted dispatch delay is:
\begin{align}
\label{eq:deltacost}
\Delta\mathrm{DelayProxy}(c) &\triangleq \mathrm{Delay}_{\mathrm{preempt}}(c) - \mathrm{Delay}_{\mathrm{wait}}(c) \nonumber \\
&= w_{\mathrm{curr}}(c) \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}}) - w_{\mathrm{wake}} \cdot R(c)
\end{align}
\end{definition}

\begin{remark}[Dimensional Consistency and Overhead Grounding]
In~\eqref{eq:deltacost}, both terms possess unified physical dimensions of $[\mathrm{weight} \times \mathrm{time}]$ (priority-weighted delay units):
\begin{itemize}
    \item $w_{\mathrm{curr}}(c) \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})$ has units $[w] \times [t]$.
    \item $w_{\mathrm{wake}} \cdot R(c)$ has units $[w] \times [t]$.
\end{itemize}
$C_{\mathrm{switch}}$ is nominally set to $50\,\mu\text{s}$ (with an effective operational sensitivity range of $20\text{--}100\,\mu\text{s}$), reflecting microarchitectural cache-line reload and TLB refill penalties on modern superscalar cores.
\end{remark}

\subsection{Online Admissible Pruning Invariants}
OPTIMA evaluates $\Delta\mathrm{DelayProxy}(c)$ across candidate cores using four pruning invariants:

\begin{enumerate}
    \item \textbf{Invariant 1 (Idle Core Trivial Minimum):} If candidate core $c$ is idle ($w_{\mathrm{curr}}(c) = 0$), then $\mathrm{Delay}_{\mathrm{preempt}}(c) = 0$ (no task is displaced). Thus:
    \begin{equation}
    \Delta\mathrm{DelayProxy}(c) = -w_{\mathrm{wake}} \cdot R(c) < 0
    \end{equation}
    Core $c$ is immediately selected for dispatch without preemption.
    
    \item \textbf{Invariant 2 (Priority Inversion Safety):} If $w_{\mathrm{curr}}(c) \ge w_{\mathrm{wake}}$, preempting core $c$ would displace an equal- or higher-priority running task to execute a lower-priority waking task, causing priority inversion. Furthermore, since $\tau_{\mathrm{slice}} + C_{\mathrm{switch}} \ge R(c)$ in typical quantum regimes, $\Delta\mathrm{DelayProxy}(c) \ge 0$. The candidate is strictly inadmissible and pruned.
    
    \item \textbf{Invariant 3 (Granularity / Anti-Thrashing Threshold):} Preempting core $c$ yields a net delay reduction ($\Delta\mathrm{DelayProxy}(c) < 0$) if and only if:
    \begin{equation}
    \label{eq:thrash_threshold}
    R(c) > \delta_{\min}(c) \triangleq \frac{w_{\mathrm{curr}}(c) \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})}{w_{\mathrm{wake}}}
    \end{equation}
    If $R(c) \le \delta_{\min}(c)$, the execution gain is insufficient to amortize displacement and context-switch penalties. The candidate is safely pruned.
    
    \item \textbf{Invariant 4 (Incumbent Greedy Cutoff):} Maintain the best incumbent delay reduction $\Delta\mathrm{Proxy}^* \triangleq \min_{j \in \mathrm{visited}} \Delta\mathrm{DelayProxy}(j)$, initialized to 0. If $\Delta\mathrm{DelayProxy}(c) \ge \Delta\mathrm{Proxy}^*$, candidate core $c$ cannot improve upon the best discovered candidate. The candidate is pruned.
\end{enumerate}

\subsection{Local One-Step Dispatch Delay Reduction}

\begin{theorem}[Local One-Step Dispatch Delay Reduction under Admissible Preemption]
\label{thm:oapf_optimality}
Let $\mathcal{P}_{\mathrm{admissible}} \subseteq \mathcal{P}$ be the set of candidate Performance cores satisfying:
\begin{enumerate}
    \item Priority superiority: $w_{\mathrm{wake}} > w_{\mathrm{curr}}(c)$;
    \item Anti-thrashing remaining runtime: $R(c) > \delta_{\min}(c)$, where
    \begin{equation}
    \delta_{\min}(c) \triangleq \frac{w_{\mathrm{curr}}(c) \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})}{w_{\mathrm{wake}}}
    \end{equation}
\end{enumerate}
If $\mathcal{P}_{\mathrm{admissible}} \ne \emptyset$, selecting core $c^* = \arg\min_{c \in \mathcal{P}_{\mathrm{admissible}}} \Delta\mathrm{DelayProxy}(c)$ and preempting $J_{\mathrm{curr}}(c^*)$ guarantees that the one-step weighted dispatch delay is strictly reduced:
\begin{equation}
\Delta\mathrm{DelayProxy}(c^*) < 0
\end{equation}
Conversely, if $\mathcal{P}_{\mathrm{admissible}} = \emptyset$ or candidate cores satisfy $R(c) \le \delta_{\min}(c)$, preemption is suppressed, bounding priority inversion to at most $\tau(c) \le \tau_{\max}$.
\end{theorem}

\begin{proof}
Under non-preemptive wait, $J_{\mathrm{wake}}$ incurs waiting delay $R(c^*)$, while $J_{\mathrm{curr}}(c^*)$ incurs 0 delay, yielding weighted delay $\mathrm{Delay}_{\mathrm{wait}} = w_{\mathrm{wake}} R(c^*)$. Under preemption, $J_{\mathrm{wake}}$ incurs 0 dispatch delay, while $J_{\mathrm{curr}}(c^*)$ incurs delay $(\tau_{\mathrm{slice}} + C_{\mathrm{switch}})$, yielding weighted delay $\mathrm{Delay}_{\mathrm{preempt}} = w_{\mathrm{curr}}(c^*) (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})$.

Computing $\Delta\mathrm{DelayProxy}(c^*) = \mathrm{Delay}_{\mathrm{preempt}} - \mathrm{Delay}_{\mathrm{wait}}$:
\begin{equation}
\Delta\mathrm{DelayProxy}(c^*) = w_{\mathrm{curr}}(c^*) (\tau_{\mathrm{slice}} + C_{\mathrm{switch}}) - w_{\mathrm{wake}} R(c^*)
\end{equation}
Since $c^* \in \mathcal{P}_{\mathrm{admissible}}$, $R(c^*) > \delta_{\min}(c^*)$, which implies $w_{\mathrm{wake}} R(c^*) > w_{\mathrm{curr}}(c^*) (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})$, ensuring $\Delta\mathrm{DelayProxy}(c^*) < 0$.

When preemption is suppressed (e.g., $w_{\mathrm{curr}}(c) \ge w_{\mathrm{wake}}$ or $R(c) \le \delta_{\min}(c)$), the waking task waits for quantum expiration. Because each executing slice is clamped to $\tau(c) \le \tau_{\max}$, the maximum waiting delay is bounded by $\tau_{\max}$, guaranteeing bounded priority inversion while protecting cache locality and pipeline execution.
\end{proof}

\begin{remark}[Tunable Anti-Thrashing Dampening Penalty $C_{\mathrm{switch}} = 50\,\mu\text{s}$]
While raw CPU register context switches require only $1\text{--}2\,\mu\text{s}$ and individual cache line reloads require tens to hundreds of nanoseconds, repeatedly displacing and reloading a memory-intensive working set introduces cumulative pipeline stalls across thousands of cycles. In OPTIMA, $C_{\mathrm{switch}}$ serves strictly as a \emph{tunable anti-thrashing dampening penalty} nominally set to $50\,\mu\text{s}$ (with an effective operational sensitivity range of $20\text{--}100\,\mu\text{s}$). Rather than modeling a single hardware cache miss, $C_{\mathrm{switch}}$ enforces an algorithmic hysteresis barrier, ensuring that preemption is triggered only when the waking task's priority advantage is sufficiently pronounced to justify evicting the incumbent thread and disrupting pipeline throughput.
\end{remark}

\subsection{Pseudocode for Online Admissible Preemption Filter}
Algorithm~\ref{alg:oapf} presents the complete, verified logic of the Online Admissible Preemption Filter.

\begin{algorithm}[t]
\caption{OPTIMA-OAPF (Online Admissible Preemption Filter)}
\label{alg:oapf}
\begin{algorithmic}[1]
\footnotesize
\REQUIRE Waking task $J_{\mathrm{wake}}$, task context $tctx$, candidate core set $\mathcal{P}$
\ENSURE Target preemption CPU $c^*$ (or $-1$ if wait is optimal)
\STATE $c^* \leftarrow -1$
\STATE $\Delta\mathrm{Proxy}^* \leftarrow 0$ \COMMENT{Incumbent threshold: must yield net delay reduction}
\STATE $now \leftarrow \text{scx\_bpf\_now}()$
\STATE $w_{\mathrm{wake}} \leftarrow \max(tctx.weight, 100)$
\STATE $\tau_{\mathrm{slice}} \leftarrow \tau_{\min}$ \COMMENT{Execution slice allocated to waking task}
\STATE $C_{\mathrm{switch}} \leftarrow 50\,\mu\text{s}$ \COMMENT{Context-switch and cache displacement overhead}

\FOR{each core $c \in \mathcal{P}$}
    \IF{$c \notin J_{\mathrm{wake}}.\text{cpus\_allowed}$}
        \STATE \textbf{continue}
    \ENDIF
    \STATE $st \leftarrow \text{lookup\_cpu\_run\_state}(c)$
    \IF{$st == \text{NULL}$ \OR $st.pid == 0$}
        \RETURN $c$ \COMMENT{Rule 1: Core is idle; trivial zero-cost dispatch}
    \ENDIF
    \IF{$st.weight \ge w_{\mathrm{wake}}$}
        \STATE \textbf{continue} \COMMENT{Rule 2: Inversion safety; candidate pruned}
    \ENDIF
    \STATE $elapsed \leftarrow now - st.start\_time$
    \STATE $R(c) \leftarrow (st.slice > elapsed) \ ? \ (st.slice - elapsed) : 0$
    \STATE $\delta_{\min} \leftarrow \dfrac{st.weight \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}})}{w_{\mathrm{wake}}}$
    \IF{$R(c) \le \delta_{\min}$}
        \STATE \textbf{continue} \COMMENT{Rule 3: Anti-thrashing; candidate pruned}
    \ENDIF
    \STATE $\Delta\mathrm{DelayProxy} \leftarrow st.weight \cdot (\tau_{\mathrm{slice}} + C_{\mathrm{switch}}) - w_{\mathrm{wake}} \cdot R(c)$
    \IF{$\Delta\mathrm{DelayProxy} < \Delta\mathrm{Proxy}^*$}
        \STATE $\Delta\mathrm{Proxy}^* \leftarrow \Delta\mathrm{DelayProxy}$
        \STATE $c^* \leftarrow c$
    \ENDIF
\ENDFOR

\IF{$c^* \ge 0$ \AND $\Delta\mathrm{Proxy}^* < 0$}
    \STATE $\text{scx\_bpf\_kick\_cpu}(c^*, \text{SCX\_KICK\_PREEMPT})$
    \RETURN $c^*$
\ENDIF
\RETURN $-1$ \COMMENT{Wait for quantum expiration is optimal}
\end{algorithmic}
\end{algorithm}

\subsection{Virtual Deadline Calculus}
To maintain density-ordered scheduling across dynamic online arrivals without global queue sorting overheads, OPTIMA maintains virtual deadlines in eBPF Dispatch Queues (DSQs).

\begin{definition}[Virtual Deadline Update Equation]
\label{def:vdeadline}
When task $J_j$ is enqueued into dispatch queue $k$ at time $t$, its virtual deadline $v_j(t)$ is computed via:
\begin{equation}
\label{eq:vdeadline}
v_j(t) = \max\Big( v_j(t^-), \, t \Big) + \Delta v_j(t)
\end{equation}
where $\Delta v_j(t)$ is the execution-density-scaled virtual deadline advance:
\begin{equation}
\label{eq:delta_v_formula}
\Delta v_j(t) = \mathrm{clamp}\left( \tau_j(t) \cdot \frac{\rho_{\mathrm{ref}}}{\rho_j}, \; \tau_{\min}, \; \tau_{\max} \right)
\end{equation}
with task density $\rho_j = \frac{\max(w_j, 10)}{p_{\mathrm{est},j}}$ and reference density $\rho_{\mathrm{ref}} = \frac{w_{\mathrm{ref}}}{p_{\mathrm{ref}}} = \frac{100}{1000\,\mu\text{s}} = 0.1\,\mu\text{s}^{-1}$. In integer arithmetic:
\begin{equation}
\Delta v_j(t) = \mathrm{clamp}\left( \frac{\tau_j(t) \cdot p_{\mathrm{est},j}}{10 \cdot \max(w_j, 10)}, \; \tau_{\min}, \; \tau_{\max} \right)
\end{equation}
Here $\tau_j(t)$ is the adaptive scheduling quantum computed as:
\begin{equation}
\label{eq:adaptive_slice}
\tau_j(t) = \mathrm{clamp}\left( \frac{Q_{\mathrm{base}}}{1 + q_k(t)}, \; \tau_{\min}, \; \tau_{\max} \right)
\end{equation}
where $q_k(t)$ denotes current queue depth of target queue $k$, $Q_{\mathrm{base}}$ is base nominal quantum, and $\tau_{\min}, \tau_{\max}$ are hardware enforcement boundaries.
\end{definition}

\begin{remark}[Decoupling Domain Placement from Intra-Queue Virtual-Deadline Ordering]
\label{rem:domain_vs_density_theory}
We explicitly decouple coarse-grained domain placement from fine-grained intra-queue scheduling:
\begin{enumerate}
    \item \textbf{Domain Placement (Algorithm~\ref{alg:enqueue}):} Tasks are routed to target DSQs using simple heuristic thresholds: tasks with $w_j \ge w_{\mathrm{RT}}$ route to \texttt{DSQ\_PERF}, long-running batch jobs ($p_{\mathrm{est},j} \ge 3\,\mathrm{ms}$ or \texttt{is\_batch}) route to \texttt{DSQ\_EFF}, and general interactive tasks route to domestic shared queues (\texttt{DSQ\_SHARED\_P} or \texttt{DSQ\_SHARED\_E}) preserving CPU stickiness. Domain routing does not evaluate $\rho = w/p$ directly.
    \item \textbf{Intra-Queue Scheduling (Algorithm~\ref{alg:enqueue}):} Task density $\rho_j$ and reference density $\rho_{\mathrm{ref}}$ are evaluated exclusively within the virtual deadline calculus ($\Delta v_j \propto \rho_{\mathrm{ref}}/\rho_j$) to establish the execution order ($\le_v$) within shared dispatch queues, prioritizing higher-density tasks ahead of lower-density competitors.
\end{enumerate}
\end{remark}

For interactive tasks ($w_j \ge w_{\mathrm{RT}}$ or $p_j < \tau_{\min}$), $v_j(t) = t$, placing them at the front of the DSQ priority heap. For batch tasks, virtual deadlines advance monotonically in proportion to quantum consumed scaled inversely by execution density ($\Delta v_j \propto 1/\rho_j = p_{\mathrm{est},j}/w_j$), maintaining Generalized Processor Sharing (GPS) fairness while approximating Smith's density ordering.

\subsection{Queue State Dynamics}
We model the system across four kernel dispatch queues:
\begin{align}
\mathcal{K} &\triangleq \{1, 2, 3, 4\} \nonumber \\
&\equiv \{\mathrm{DSQ\_PERF}, \, \mathrm{DSQ\_SHARED\_P}, \nonumber \\
&\quad\quad \mathrm{DSQ\_SHARED\_E}, \, \mathrm{DSQ\_EFF}\}
\end{align}
Let $\mathbf{Q}(t) = \big( q_1(t), q_2(t), q_3(t), q_4(t) \big)^T \in \mathbb{Z}_{\ge 0}^4$ represent the queue backlog vector at discrete time slot $t \in \mathbb{Z}_{\ge 0}$.
The discrete-time queue state evolution equation is:
\begin{equation}
\label{eq:queue_evol}
q_k(t+1) = \max\Big(0, \; q_k(t) - s_k(t)\Big) + a_k(t), \quad \forall k \in \mathcal{K}
\end{equation}
where:
\begin{itemize}
    \item $a_k(t)$ is the number of task arrivals into queue $k$ during time slot $t$, with arrival rate $\lambda_k \triangleq \mathbb{E}[a_k(t)]$ and bounded second moment $\mathbb{E}[a_k(t)^2] \le a_{\max}^2 < \infty$.
    \item $s_k(t)$ is the service capacity (number of task dispatches plus work-stealing removals) from queue $k$ during time slot $t$, with $s_k(t) \le s_{\max} < \infty$.
\end{itemize}

\subsection{Quadratic Lyapunov Function and Drift Analysis}
\begin{definition}[Quadratic Lyapunov Function]
We define the quadratic Lyapunov function $L(\mathbf{Q}(t))$ as:
\begin{equation}
\label{eq:lyapunov_func}
L(\mathbf{Q}(t)) \triangleq \frac{1}{2} \sum_{k=1}^4 q_k(t)^2 = \frac{1}{2} \|\mathbf{Q}(t)\|_2^2
\end{equation}
\end{definition}

The conditional one-step Lyapunov drift is defined as:
\begin{equation}
\Delta L(\mathbf{Q}(t)) \triangleq \mathbb{E}\Big[ L(\mathbf{Q}(t+1)) - L(\mathbf{Q}(t)) \;\Big|\; \mathbf{Q}(t) \Big]
\end{equation}

\begin{lemma}[Exact Algebraic Lyapunov Drift Upper Bound]
\label{lem:drift_bound}
For the queue dynamics in~\eqref{eq:queue_evol}, the one-step Lyapunov drift satisfies:
\begin{equation}
\label{eq:drift_bound}
\Delta L(\mathbf{Q}(t)) \le B - \sum_{k=1}^4 q_k(t) \, \mathbb{E}\Big[ s_k(t) - a_k(t) \;\Big|\; \mathbf{Q}(t) \Big]
\end{equation}
where $B$ is a finite positive constant defined as:
\begin{equation}
\label{eq:B_def}
B \triangleq \frac{1}{2} \sum_{k=1}^4 \mathbb{E}\Big[ a_k(t)^2 + s_k(t)^2 \;\Big|\; \mathbf{Q}(t) \Big] \le 2(a_{\max}^2 + s_{\max}^2) < \infty
\end{equation}
\end{lemma}

\begin{proof}
From the queue evolution equation~\eqref{eq:queue_evol}, for each queue $k$:
\begin{equation}
q_k(t+1) = \max\Big(0, \; q_k(t) - s_k(t)\Big) + a_k(t)
\end{equation}
Squaring both sides:
\begin{align}
\label{eq:expand_sq}
q_k(t+1)^2 &= \left( \max\Big(0, \; q_k(t) - s_k(t)\Big) + a_k(t) \right)^2 \nonumber \\
&= \left(\max\Big(0, \; q_k(t) - s_k(t)\Big)\right)^2 + a_k(t)^2 \nonumber \\
&\quad + 2 a_k(t) \max\Big(0, \; q_k(t) - s_k(t)\Big)
\end{align}

We now bound each component of~\eqref{eq:expand_sq}:
\begin{enumerate}
    \item For any real number $x$, $(\max(0, x))^2 \le x^2$. Setting $x = q_k(t) - s_k(t)$:
    \begin{equation}
    \label{eq:sq_bound}
    \left(\max\Big(0, \; q_k(t) - s_k(t)\Big)\right)^2 \le \big( q_k(t) - s_k(t) \big)^2
    \end{equation}
    Expanding the square:
    \begin{equation}
    \big( q_k(t) - s_k(t) \big)^2 = q_k(t)^2 - 2 q_k(t) s_k(t) + s_k(t)^2
    \end{equation}
    
    \item For the cross-term, since $s_k(t) \ge 0$ and $q_k(t) \ge 0$, we have $q_k(t) - s_k(t) \le q_k(t)$, which implies:
    \begin{equation}
    \max\Big(0, \; q_k(t) - s_k(t)\Big) \le q_k(t)
    \end{equation}
    Because $a_k(t) \ge 0$, multiplying both sides by $2 a_k(t)$ yields:
    \begin{align}
    \label{eq:cross_bound}
    2 a_k(t) \max\Big(0, \; q_k(t) - s_k(t)\Big) &\le 2 a_k(t) q_k(t) \nonumber \\
    &= 2 q_k(t) a_k(t)
    \end{align}
\end{enumerate}

Substituting~\eqref{eq:sq_bound} and~\eqref{eq:cross_bound} back into~\eqref{eq:expand_sq}:
\begin{align}
q_k(t+1)^2 &\le q_k(t)^2 - 2 q_k(t) s_k(t) + s_k(t)^2 + a_k(t)^2 \nonumber \\
&\quad + 2 q_k(t) a_k(t) \nonumber \\
&= q_k(t)^2 + a_k(t)^2 + s_k(t)^2 \nonumber \\
&\quad - 2 q_k(t) \big( s_k(t) - a_k(t) \big)
\end{align}

Subtracting $q_k(t)^2$ from both sides and dividing by 2:
\begin{align}
\frac{1}{2} \Big( q_k(t+1)^2 - q_k(t)^2 \Big) &\le \frac{1}{2}\big( a_k(t)^2 + s_k(t)^2 \big) \nonumber \\
&\quad - q_k(t)\big( s_k(t) - a_k(t) \big)
\end{align}

Summing over all four queues $k \in \{1, 2, 3, 4\}$ and taking conditional expectations given $\mathbf{Q}(t)$:
\begin{align}
\Delta L(\mathbf{Q}(t)) &\le \sum_{k=1}^4 \frac{1}{2}\mathbb{E}\big[ a_k(t)^2 + s_k(t)^2 \mid \mathbf{Q}(t) \big] \nonumber \\
&\quad - \sum_{k=1}^4 q_k(t) \mathbb{E}\big[ s_k(t) - a_k(t) \mid \mathbf{Q}(t) \big] \nonumber \\
&= B - \sum_{k=1}^4 q_k(t) \mathbb{E}\big[ s_k(t) - a_k(t) \mid \mathbf{Q}(t) \big]
\end{align}
Because arrival variances and service capacities per scheduling time slot are physically bounded by finite dispatch limits ($a_k(t) \le a_{\max} < \infty$, $s_k(t) \le s_{\max} < \infty$), the constant $B = \frac{1}{2}\sum_{k=1}^4 \mathbb{E}[a_k(t)^2 + s_k(t)^2 \mid \mathbf{Q}(t)]$ is strictly finite.
\end{proof}

\subsection{Workload Admissibility, Queue Stability, and Starvation Freedom}

\begin{definition}[Decoupled Class-Admissible Workload Region]
\label{def:class_admissible}
The arrival workload is class-admissible if:
\begin{enumerate}
    \item Real-time demand to $\texttt{DSQ\_PERF}$ ($q_1$) is sub-capacity on the Performance partition:
    \begin{equation}
    \lambda_1 \bar{p}_1 \le M_P c_P - \epsilon_1, \quad \text{for some } \epsilon_1 > 0
    \end{equation}
    \item Aggregate workload demand is strictly within total multiprocessor capacity:
    \begin{equation}
    \label{eq:capacity_constraint}
    \sum_{k=1}^4 \lambda_k \bar{p}_k \le C_{\mathrm{total}} - \epsilon_0, \quad \epsilon_0 > 0
    \end{equation}
\end{enumerate}
where $C_{\mathrm{total}} = M_P c_P + M_E c_E$, $\lambda_k = \mathbb{E}[a_k(t)]$, and $\bar{p}_k$ is the mean nominal execution requirement of tasks arriving in queue $k$.
\end{definition}

\begin{theorem}[Conditional Multi-Queue Stability via Foster-Lyapunov Drift]
\label{thm:stability}
Let the multi-queue backlog $\mathbf{Q}(t)$ evolve under an aggregate-admissible workload (Definition~\ref{def:class_admissible}). If the dispatch and hierarchical work-stealing policy guarantees a positive service slack across all active non-empty queues:
\begin{equation}
\label{eq:slack_cond}
\mathbb{E}\big[ s_k(t) - a_k(t) \;\big|\; \mathbf{Q}(t) \big] \ge \delta > 0, \quad \forall k: q_k(t) > 0
\end{equation}
then the quadratic Lyapunov function satisfies the negative drift condition:
\begin{equation}
\label{eq:negative_drift}
\Delta L(\mathbf{Q}(t)) \le B - \delta \sum_{k=1}^4 q_k(t)
\end{equation}
for finite positive constant $B = \frac{1}{2} \sum_{k=1}^4 \mathbb{E}[a_k(t)^2 + s_k(t)^2 \mid \mathbf{Q}(t)] < \infty$. Consequently, the queue backlog process $\mathbf{Q}(t)$ is Foster-Lyapunov stable (positive recurrent), with bounded time-average queue depth:
\begin{equation}
\limsup_{T \to \infty} \frac{1}{T} \sum_{t=0}^{T-1} \sum_{k=1}^4 \mathbb{E}[q_k(t)] \le \frac{B}{\delta} < \infty
\end{equation}
\end{theorem}

\begin{proof}
From Lemma~\ref{lem:drift_bound}, the one-step Lyapunov drift satisfies:
\begin{equation}
\Delta L(\mathbf{Q}(t)) \le B - \sum_{k=1}^4 q_k(t) \, \mathbb{E}\big[ s_k(t) - a_k(t) \;\big|\; \mathbf{Q}(t) \big]
\end{equation}
Under the stated conditional per-queue service slack condition~\eqref{eq:slack_cond}, $\mathbb{E}[s_k(t) - a_k(t) \mid \mathbf{Q}(t)] \ge \delta > 0$ for all queues with $q_k(t) > 0$. Substituting this directly into the drift bound yields:
\begin{equation}
\Delta L(\mathbf{Q}(t)) \le B - \sum_{k=1}^4 q_k(t) \cdot \delta = B - \delta \sum_{k=1}^4 q_k(t)
\end{equation}

By the Foster-Lyapunov Stability Criterion (Neveu's Theorem for irreducible Markov chains), the negative drift condition implies that the queue state process $\mathbf{Q}(t)$ is positive recurrent.
Summing~\eqref{eq:negative_drift} over time slots $t = 0, 1, \dots, T-1$:
\begin{equation}
\mathbb{E}[L(\mathbf{Q}(T))] - \mathbb{E}[L(\mathbf{Q}(0))] \le T B - \delta \sum_{t=0}^{T-1} \sum_{k=1}^4 \mathbb{E}[q_k(t)]
\end{equation}
Since $L(\mathbf{Q}(T)) \ge 0$:
\begin{equation}
\frac{1}{T} \sum_{t=0}^{T-1} \sum_{k=1}^4 \mathbb{E}[q_k(t)] \le \frac{B}{\delta} + \frac{\mathbb{E}[L(\mathbf{Q}(0))]}{\delta T}
\end{equation}
Taking the limit superior as $T \to \infty$ establishes strong aggregate queue stability:
\begin{equation}
\limsup_{T \to \infty} \frac{1}{T} \sum_{t=0}^{T-1} \sum_{k=1}^4 \mathbb{E}[q_k(t)] \le \frac{B}{\delta} < \infty
\end{equation}
\end{proof}

\begin{remark}[Architectural Realization of the Service Slack Condition]
In multi-queue systems with strict priorities, global capacity admissibility ($\sum \lambda_k \bar{p}_k \le C_{\mathrm{total}} - \epsilon_0$) is a necessary condition for aggregate stability, but does not independently ensure per-queue service slack~\eqref{eq:slack_cond} if high-priority queues can monopolize all cores. OPTIMA establishes this per-queue condition through structural partitioning:
(1) dedicated Efficiency cores provide guaranteed domestic compute capacity to batch queues (\texttt{DSQ\_EFF}), preventing high-priority P-core traffic from completely starving throughput work; and
(2) dynamic quantum scaling $\tau(q_k) = \max(\frac{Q_{\mathrm{base}}}{1+q_k}, \tau_{\min})$ dynamically compresses time slices during congestion, accelerating task dispatch and preemption frequency.
\end{remark}

\begin{proposition}[Virtual Deadline Bounding and Starvation Protection]
\label{prop:starvation}
Under OPTIMA's virtual deadline calculus (Algorithm~\ref{alg:enqueue}), for every admitted task $J_j$ enqueued into a batch or interactive queue (\texttt{DSQ\_SHARED\_P}, \texttt{DSQ\_SHARED\_E}, \texttt{DSQ\_EFF}) at time $t$, its assigned virtual deadline satisfies:
\begin{equation}
v_j(t) = \operatorname{clamp}(v_{\mathrm{time}}, t, t + 2\tau_j) + \Delta v_j \le t + 3\tau_{\max}
\end{equation}
since $\tau_j \le \tau_{\max}$ and $\Delta v_j \le \tau_{\max}$. Any subsequent batch task $J_{j'}$ arriving at timestamp $t' > t + 3\tau_{\max}$ receives $v_{j'}(t') \ge t' > v_j(t)$ and cannot be scheduled ahead of $J_j$. While a batch task arriving earlier ($t' \le t + 3\tau_{\max}$) with higher density may receive an earlier deadline, $v_j(t) \le t + 3\tau_{\max}$ strictly bounds relative queue advancement lag. Under the conditional service rate condition ($\mathbb{E}[s_k - a_k \mid \mathbf{Q}] \ge \delta > 0$) of Theorem~\ref{thm:stability}, every admitted batch task is guaranteed dispatch within finite time. Furthermore, latency-critical tasks ($w \ge w_{\mathrm{RT}}$) receive immediate priority ($v_j = t$) and always supersede batch tasks; when preemption is suppressed by OAPF, the immediate waiting delay imposed by the running task is bounded by $\tau_{\max}$, ensuring bounded priority inversion.
\end{proposition}

\section{Algorithmic Specifications}
\label{sec:algorithms}

We formalize OPTIMA's core scheduling operations implemented in Linux \texttt{sched\_ext} eBPF: CPU selection, queue insertion, and work-conserving dispatch.

% -------------------------------------------------------------------------
% ALGORITHM 1
% -------------------------------------------------------------------------
\begin{algorithm*}[t]
\caption{OPTIMA-SELECT-CPU (Synchronous IPC Fast-Path \& Asymmetric Domain Selection)}
\label{alg:select_cpu}
\begin{algorithmic}[1]
\footnotesize
\REQUIRE Task $p$, previous CPU $prev\_cpu$, wakeup flags $wake\_flags$
\ENSURE Selected target CPU $target\_cpu$
\STATE $tctx \leftarrow \text{lookup\_task\_ctx}(p)$
\IF{$p$ is pinned OR $p$ is kernel thread}
    \STATE Dispatch directly to $prev\_cpu$ via \texttt{SCX\_DSQ\_LOCAL}; \textbf{return} $prev\_cpu$
\ENDIF
\STATE $this\_cpu \leftarrow \text{smp\_processor\_id}()$, \quad $idle\_smtmask \leftarrow \text{scx\_bpf\_get\_idle\_smtmask}()$, \quad $prev\_is\_perf \leftarrow \text{is\_perf\_core}(prev\_cpu)$
\IF{$wake\_flags \ \& \ \text{SCX\_WAKE\_SYNC}$}
    \IF{$this\_cpu \in p.\text{cpus\_allowed}$ \AND $\text{dsq\_nr\_queued}(this\_cpu) == 0$}
        \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow this\_cpu$
    \ENDIF
    \STATE $this\_sib \leftarrow \text{get\_smt\_sibling}(this\_cpu)$
    \IF{$this\_sib \ge 0$ \AND $this\_sib \in p.\text{cpus\_allowed}$ \AND $\text{dsq\_nr\_queued}(this\_sib) == 0$}
        \STATE $\text{test\_and\_clear\_idle}(this\_sib)$; \quad \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow this\_sib$
    \ENDIF
    \IF{$prev\_cpu \in p.\text{cpus\_allowed}$ \AND $\text{test\_and\_clear\_idle}(prev\_cpu)$}
        \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow prev\_cpu$
    \ENDIF
\ENDIF
\IF{$prev\_is\_perf$ \AND $prev\_cpu \in p.\text{cpus\_allowed}$ \AND $prev\_cpu \in idle\_smtmask$ \AND $\text{test\_and\_clear\_idle}(prev\_cpu)$}
    \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow prev\_cpu$
\ENDIF
\IF{$tctx.core\_type \ne \text{CORE\_TYPE\_EFF}$}
    \STATE $c \leftarrow \text{pick\_idle\_domestic\_core}(p, \text{is\_perf}=\text{TRUE}, idle\_smtmask)$
    \IF{$c \ge 0$}
        \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow c$
    \ENDIF
    \STATE $c \leftarrow \text{pick\_idle\_domestic\_cpu}(p, \text{is\_perf}=\text{TRUE})$
    \IF{$c \ge 0$}
        \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow c$
    \ENDIF
\ENDIF
\IF{$prev\_is\_perf$ \AND $p \in p.\text{cpus\_allowed}$}
    \IF{$\text{dsq\_nr\_queued}(\text{DSQ\_SHARED\_P}) > 0$ \OR $(tctx.is\_batch \land tctx.avg\_runtime \ge 3\text{ms})$}
        \STATE $e \leftarrow \text{pick\_idle\_domestic\_core}(p, \text{is\_perf}=\text{FALSE}, idle\_smtmask)$
        \IF{$e \ge 0$}
            \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow e$
        \ENDIF
    \ENDIF
    \STATE $tctx.dispatch\_local \leftarrow \text{FALSE}$; \textbf{return} $prev\_cpu$
\ENDIF
\STATE $e \leftarrow \text{pick\_idle\_domestic\_core}(p, \text{is\_perf}=\text{FALSE}, idle\_smtmask)$
\IF{$e \ge 0$}
    \STATE \textbf{goto} direct\_dispatch with $target\_cpu \leftarrow e$
\ENDIF
\IF{$tctx.weight \ge w_{\mathrm{RT}}$}
    \STATE $preempt\_cpu \leftarrow \text{OPTIMA-OAPF}(p, tctx, \mathcal{P})$
    \IF{$preempt\_cpu \ge 0$}
        \STATE $tctx.dispatch\_local \leftarrow \text{TRUE}$; \textbf{return} $preempt\_cpu$
    \ENDIF
\ENDIF
\STATE $tctx.dispatch\_local \leftarrow \text{FALSE}$; \textbf{return} $prev\_cpu$
\STATE \textbf{direct\_dispatch:}
\STATE Compute slice $\tau \in [\tau_{\min}, \tau_{\max}]$ based on $tctx.is\_batch$ and $tctx.avg\_runtime$
\STATE $\text{scx\_bpf\_dsq\_insert}(p, \text{SCX\_DSQ\_LOCAL}, \tau, 0)$
\RETURN $target\_cpu$
\end{algorithmic}
\end{algorithm*}

% -------------------------------------------------------------------------
% ALGORITHM 2 (ENQUEUE)
% -------------------------------------------------------------------------
\begin{algorithm}[t]
\caption{OPTIMA-ENQUEUE (Virtual Deadline Calculus \& OAPF Preemption Kick)}
\label{alg:enqueue}
\begin{algorithmic}[1]
\footnotesize
\REQUIRE Task $p$, enqueue flags $enq\_flags$
\STATE $tctx \leftarrow \text{lookup\_task\_ctx}(p)$
\STATE $task\_cpu \leftarrow \text{scx\_bpf\_task\_cpu}(p)$
\IF{$p$ is pinned OR $p$ is kernel thread OR $tctx.dispatch\_local$}
    \STATE $tctx.dispatch\_local \leftarrow \text{FALSE}$
    \STATE $\text{scx\_bpf\_dsq\_insert}(p, \text{SCX\_DSQ\_LOCAL}, \tau_{\min}, enq\_flags)$; \textbf{return}
\ENDIF
\STATE \COMMENT{Density-Guided Domain Placement Target DSQ Selection}
\IF{$tctx.weight \ge w_{\mathrm{RT}}$}
    \STATE $target\_dsq \leftarrow \text{DSQ\_PERF}$
\ELSIF{$tctx.avg\_runtime \ge 3\text{ms} \lor tctx.is\_batch$}
    \STATE $target\_dsq \leftarrow \text{DSQ\_EFF}$
\ELSE
    \STATE $target\_dsq \leftarrow \text{is\_perf\_core}(task\_cpu) \ ? \ \text{DSQ\_SHARED\_P} : \text{DSQ\_SHARED\_E}$
\ENDIF
\STATE $is\_rt \leftarrow (tctx.weight \ge w_{\mathrm{RT}} \lor (\neg tctx.is\_batch \land tctx.avg\_runtime < 1\text{ms}))$
\STATE $now \leftarrow \text{scx\_bpf\_now}()$
\IF{$is\_rt$}
    \STATE $\tau \leftarrow \tau_{\min}$, \quad $deadline \leftarrow now$
\ELSE
    \IF{$tctx.avg\_runtime \ge 3\text{ms}$}
        \STATE $\tau \leftarrow \tau_{\max}$
    \ELSE
        \STATE $q \leftarrow \text{scx\_bpf\_dsq\_nr\_queued}(target\_dsq)$
        \STATE $\tau \leftarrow \max\left( \dfrac{Q_{\mathrm{base}}}{1 + q}, \; \tau_{\min} \right)$
    \ENDIF
    \STATE $v \leftarrow \mathrm{clamp}(p.\text{scx.dsq\_vtime}, now, now + 2\tau)$
    \STATE $p_{\mathrm{est}} \leftarrow \max(tctx.avg\_runtime, \tau_{\min})$
    \STATE $\rho \leftarrow \dfrac{\max(tctx.weight, 10)}{p_{\mathrm{est}}}$ \COMMENT{Execution density $\rho = w / p_{\mathrm{est}}$}
    \STATE $\Delta v \leftarrow \mathrm{clamp}\left( \dfrac{\tau \cdot p_{\mathrm{est}}}{10 \cdot \max(tctx.weight, 10)}, \; \tau_{\min}, \; \tau_{\max} \right)$ \COMMENT{Monotonic increment $\tau \cdot \rho_{\mathrm{ref}} / \rho$}
    \STATE $deadline \leftarrow v + \Delta v$
\ENDIF
\STATE $p.\text{scx.dsq\_vtime} \leftarrow deadline$
\STATE $\text{scx\_bpf\_dsq\_insert\_vtime}(p, target\_dsq, \tau, deadline, enq\_flags)$
\STATE \COMMENT{Online Admissible Preemption Kick}
\STATE $idle\_cpu \leftarrow \text{scx\_bpf\_pick\_idle\_cpu}(p.\text{cpus\_allowed}, 0)$
\IF{$idle\_cpu \ge 0$}
    \STATE $\text{scx\_bpf\_kick\_cpu}(idle\_cpu, \text{SCX\_KICK\_IDLE})$; \textbf{return}
\ENDIF
\IF{$is\_rt$}
    \STATE $preempt\_cpu \leftarrow \text{OPTIMA-OAPF}(p, tctx, \mathcal{P})$
    \IF{$preempt\_cpu \ge 0$}
        \STATE $\text{scx\_bpf\_kick\_cpu}(preempt\_cpu, \text{SCX\_KICK\_PREEMPT})$
    \ENDIF
\ENDIF
\end{algorithmic}
\end{algorithm}

% -------------------------------------------------------------------------
% ALGORITHM 3 (DISPATCH)
% -------------------------------------------------------------------------
\begin{algorithm}[t]
\caption{OPTIMA-DISPATCH (Asymmetric Hierarchical Work-Stealing)}
\label{alg:dispatch}
\begin{algorithmic}[1]
\footnotesize
\REQUIRE Executing core $cpu$, previous task $prev$
\STATE $is\_perf \leftarrow \text{is\_perf\_core}(cpu)$
\IF{$is\_perf$}
    \STATE \COMMENT{Priority Tier 1: Latency-critical tasks}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_PERF}, 0)$}
        \RETURN
    \ENDIF
    \STATE \COMMENT{Priority Tier 2: Domestic P-core shared queue}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_SHARED\_P}, 0)$}
        \RETURN
    \ENDIF
    \STATE \COMMENT{Priority Tier 3: Cross-domain work-stealing from E-domain}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_SHARED\_E}, 0)$}
        \RETURN
    \ENDIF
    \STATE \COMMENT{Priority Tier 4: Throughput steal from batch queue}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_EFF}, 0)$}
        \RETURN
    \ENDIF
\ELSE
    \STATE \COMMENT{Priority Tier 1: Domestic E-core shared queue}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_SHARED\_E}, 0)$}
        \RETURN
    \ENDIF
    \STATE \COMMENT{Priority Tier 2: Throughput batch queue}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_EFF}, 0)$}
        \RETURN
    \ENDIF
    \STATE \COMMENT{Priority Tier 3: Cross-domain steal from P-domain}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_SHARED\_P}, 0)$}
        \RETURN
    \ENDIF
    \STATE \COMMENT{Priority Tier 4: Work-conserving safety steal from RT queue}
    \IF{$\text{scx\_bpf\_dsq\_move\_to\_local}(\text{DSQ\_PERF}, 0)$}
        \RETURN
    \ENDIF
\ENDIF
\end{algorithmic}
\end{algorithm}
